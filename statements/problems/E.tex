\begin{problem}{E}{Hello}{1 second}

\noindent
Hello World

\InputFile

\noindent
Hello World

\OutputFile

\noindent
Hello World

\Examples

\begin{example}
\exmp{1}{
1234
}{
Hello World
}
\end{example}
\Explaination{\upshape\normalsize The glass diagonals cut the interior of the $5 \times 6$ pocket into many separate regions, as drawn below. Breaking the glass in $13$ suitably chosen cells is enough to connect them all into one region, and no smaller set of breaks works.}

\begin{center}
\begin{tikzpicture}[scale=0.62, line join=round]
  % cell grid lines (open cell boundaries)
  \draw[gray!40] (1,0)--(1,5) (2,0)--(2,5) (3,0)--(3,5) (4,0)--(4,5) (5,0)--(5,5);
  \draw[gray!40] (0,1)--(6,1) (0,2)--(6,2) (0,3)--(6,3) (0,4)--(6,4);
  % outer border of the pocket
  \draw[very thick] (0,0) rectangle (6,5);
  % glass diagonals, transcribed row by row from the input
  \draw[thick,blue!70!black]
    (0,4)--(1,5) (1,5)--(2,4) (2,4)--(3,5) (3,4)--(4,5) (4,5)--(5,4) (5,4)--(6,5) % row 1: /\//\/
    (0,4)--(1,3) (1,3)--(2,4) (2,4)--(3,3) (3,3)--(4,4) (4,4)--(5,3) (5,3)--(6,4) % row 2: \/\/\/
    (0,2)--(1,3) (1,3)--(2,2) (2,2)--(3,3) (3,2)--(4,3) (4,3)--(5,2) (5,2)--(6,3) % row 3: /\//\/
    (0,2)--(1,1) (1,1)--(2,2) (2,2)--(3,1) (3,1)--(4,2) (4,2)--(5,1) (5,1)--(6,2) % row 4: \/\/\/
    (0,1)--(1,0) (1,0)--(2,1) (2,1)--(3,0) (3,0)--(4,1) (4,1)--(5,0) (5,0)--(6,1);% row 5: \/\/\/
\end{tikzpicture}
\\[2pt]
{\itshape Sample 1: the $5 \times 6$ glass layout.}
\end{center}

\begin{example}
\exmp{2}{
2 2
\textbackslash/
/\textbackslash
}{
  3
}
\end{example}
\Explaination{\upshape\normalsize Here the four diagonals meet at the centre and cut the $2 \times 2$ pocket into the four regions shown below. For instance, breaking the glass in the top-left, bottom-left, and bottom-right cells connects all of them into one region, so $3$ operations suffice; no fewer than $3$ can work.}

\begin{center}
\begin{tikzpicture}[scale=1.6, line join=round]
  % the four regions, coloured
  \fill[blue!12]   (0,2)--(2,2)--(1,1)--cycle; % top
  \fill[orange!18] (0,0)--(0,2)--(1,1)--cycle; % left
  \fill[green!16]  (0,0)--(2,0)--(1,1)--cycle; % bottom
  \fill[red!12]    (2,0)--(2,2)--(1,1)--cycle; % right
  % open cell boundaries
  \draw[gray!55,dashed] (1,0)--(1,2) (0,1)--(2,1);
  % outer border
  \draw[very thick] (0,0) rectangle (2,2);
  % glass diagonals (the two diagonals meet at the centre)
  \draw[thick,blue!70!black] (0,0)--(2,2) (0,2)--(2,0);
  % region labels
  \node at (1,1.58) {\small 1};
  \node at (0.42,1) {\small 2};
  \node at (1,0.42) {\small 3};
  \node at (1.58,1) {\small 4};
\end{tikzpicture}
\\[2pt]
{\itshape Sample 2: the glass forms an ``X'' that cuts the pocket into four regions.}
\end{center}
